113class RegexFilterStrategy : public IFilterStrategy {
114public:
115 RegexFilterStrategy() = default;
116
117 // Each instance owns its own cache (see m_cacheMutex below), so a copy
118 // must not share it — start the clone with an empty, freshly-built cache
119 // rather than copying mutex/regex state.
120 RegexFilterStrategy(const RegexFilterStrategy&) noexcept {}
121 RegexFilterStrategy& operator=(const RegexFilterStrategy&) noexcept { return *this; }
122
123 [[nodiscard]] bool matches(const std::string& value,
124 const std::string& pattern,
125 bool caseSensitive) const override;
126
127 bool isValidPattern(const std::string& pattern) const override;
128
129 std::string getName() const override { return "regex"; }
130
131 std::unique_ptr<IFilterStrategy> clone() const override {
132 return std::make_unique<RegexFilterStrategy>(*this);
133 }
134
135private:
136 // A Filter/FilterCondition owns its strategy exclusively (cloned, never
137 // shared), and the same (pattern, caseSensitive) is matched against many
138 // values in a row during one filter pass — cache the last compiled regex
139 // per instance instead of a global cache, avoiding both cross-filter
140 // lock contention and a cache-key string allocation on every call.
141 mutable std::mutex m_cacheMutex;
142 mutable std::string m_cachedPattern;
143 mutable bool m_cachedCaseSensitive {false};
144 mutable bool m_hasCachedRegex {false};
145 mutable std::regex m_cachedRegex;
146};
